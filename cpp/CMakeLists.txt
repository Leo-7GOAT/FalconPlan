cmake_minimum_required(VERSION 3.16)
project(falcon_core VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FALCON_SANITIZE "Enable ASan and UBSan" OFF)
add_library(falcon_core src/core.cpp src/planning.cpp)
set_target_properties(falcon_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(falcon_core PUBLIC include)
target_compile_options(falcon_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(FALCON_SANITIZE)
  target_compile_options(falcon_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(falcon_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
find_package(GTest QUIET)
if(NOT GTest_FOUND)
  add_subdirectory(/usr/src/googletest googletest)
endif()
add_executable(core_tests tests/test_core.cpp)
target_link_libraries(core_tests falcon_core GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(core_tests)
add_executable(core_probe src/probe.cpp)
target_link_libraries(core_probe falcon_core)
add_library(falcon_c SHARED src/c_api.cpp)
target_link_libraries(falcon_c falcon_core)
target_link_libraries(core_tests falcon_c)
